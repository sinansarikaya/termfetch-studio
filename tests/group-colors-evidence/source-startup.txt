5829: setup_shell_integration() {

5831:

5833:     if [ -n "$BASH_VERSION" ] || [ -f "$HOME/.bashrc" ]; then
5834:         # Remove any existing TermFetch Studio block
5835:         if [ -f "$HOME/.bashrc" ]; then
5836:             sed -i '/# TermFetch Studio - START/,/# TermFetch Studio - END/d' "$HOME/.bashrc" 2>/dev/null || true
5837:         fi
5838: 
5839:         # Add new integration block with markers
5840:         local external_startup=false
5841:         has_external_fetch_startup "$HOME/.bashrc" && external_startup=true
5842:         printf '\n# TermFetch Studio - START\n' >> "$HOME/.bashrc"
5843:         printf '_tfs_external_startup=%s\n' "$external_startup" >> "$HOME/.bashrc"
5844:         cat >> "$HOME/.bashrc" << 'BASHRC'
5845: 
5846: export PATH="$HOME/.local/bin:$PATH"
5847: if [[ $- == *i* ]] && [ -t 1 ] && [ "${_tfs_external_startup:-false}" = true ]; then export TFS_STARTUP_TTY="$(tty 2>/dev/null)"; fi
5848: if [[ $- == *i* ]] && [ -t 1 ] && [ "${_tfs_external_startup:-false}" != true ]; then
5849: _tfs_tty=$(tty 2>/dev/null)
5850: if [ "${TFS_STARTUP_TTY:-}" != "$_tfs_tty" ]; then
5851: export TFS_STARTUP_TTY="$_tfs_tty"
5852: [ -f ~/.config/termfetch-studio/theme.sh ] && source ~/.config/termfetch-studio/theme.sh
5853: if [ -x "$HOME/.local/bin/tfs-fastfetch" ]; then
5854:     "$HOME/.local/bin/tfs-fastfetch"
5855: elif command -v fastfetch &> /dev/null && [ -f ~/.config/termfetch-studio/fastfetch.jsonc ]; then
5856:     fastfetch --config ~/.config/termfetch-studio/fastfetch.jsonc
5857: fi
5858: fi
5859: fi
5860: unset _tfs_tty _tfs_external_startup
5861: # TermFetch Studio - END

5864:     fi

5867:         # Remove any existing TermFetch Studio block
5868:         if [ -f "$HOME/.zshrc" ]; then
5869:             sed -i '/# TermFetch Studio - START/,/# TermFetch Studio - END/d' "$HOME/.zshrc" 2>/dev/null || true
5870:         fi

5877:         cat >> "$HOME/.zshrc" << 'ZSHRC'
5878:

5880: if [[ -o interactive ]] && [ -t 1 ] && [ "${_tfs_external_startup:-false}" = true ]; then export TFS_STARTUP_TTY="$(tty 2>/dev/null)"; fi
5881: if [[ -o interactive ]] && [[ -t 1 ]] && [[ "${_tfs_external_startup:-false}" != true ]]; then
5882: _tfs_tty=$(tty 2>/dev/null)
5883: if [[ "${TFS_STARTUP_TTY:-}" != "$_tfs_tty" ]]; then

5902: 
5903:         # Add new integration block with markers
5904:         local external_startup=false

5906:         printf '\n# TermFetch Studio - START\n' >> "$HOME/.config/fish/config.fish"
5907:         printf 'set -l _tfs_external_startup %s\n' "$external_startup" >> "$HOME/.config/fish/config.fish"
